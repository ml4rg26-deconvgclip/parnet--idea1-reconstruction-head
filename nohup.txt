## ─────────────────────────────────────────────────────────────────────────
## Experimente: positionsabhaengiges Alpha + CNN-only Ablation
## ─────────────────────────────────────────────────────────────────────────

# Kontext:
# - QLayer v1 hatte einen Phasen-Init-Bug (alle phi=0 -> exakter
#   Sattelpunkt, kein Gradient). Gefixt in QLayer.__init__ (model.py):
#   phase = nn.Parameter(torch.randn(num_rbps) * 0.1) statt torch.zeros(...).
# - Nach dem Fix (QLayer v2) blieb die Performance praktisch gleich
#   (r=0.409 -> 0.407). D.h. der Vorsprung von QLayer ggue. Standard kommt
#   vermutlich vom dilated CNN, nicht von der Phasen-Interferenz.
# - Neues Flag --positional-alpha (in train_standard.py und train_qlayer.py):
#   MixCoeffHead gibt pro Position statt einmal pro Sequenz ein
#   Mischgewicht aus (B,223,L) statt (B,223) -- groesster erwarteter Hebel.
# - Neues Skript train_cnn_only.py: Standard-Mixing direkt in denselben
#   dilated CNN wie QLayer, aber OHNE Interferenz/Quadrieren dazwischen
#   -- isoliert, ob der CNN allein fuer den QLayer-Vorsprung reicht.
# - Neues Skript evaluate_new_models.py: wertet jedes der drei Modelle
#   EINZELN aus (anders als export_results_data.py, das immer Standard+
#   QLayer zusammen laedt) und schreibt getrennt nach results/new/<model>/.

# Alle drei Modelle nacheinander trainieren, danach automatisch auswerten:

nohup bash -c '
export PIXI_CACHE_DIR="$HOME/.pixi-cache"
cd /mnt/storage1/workspace/pgoldemund/parnet--idea1-reconstruction-head

echo "=== [1/4] Training: Standard + positional alpha ==="
pixi run -e parnet-dev-cu12 python scripts/train_standard.py --lambda-alpha 0.1 --lambda-nll 0.1 --positional-alpha --run-id globalclip.standard.positional_v1 --gpu 0

echo "=== [2/4] Training: QLayer + positional alpha ==="
pixi run -e parnet-dev-cu12 python scripts/train_qlayer.py --lambda-alpha 0.1 --lambda-nll 0.1 --positional-alpha --run-id globalclip.qlayer.positional_v1 --gpu 0

echo "=== [3/4] Training: CNN-only Ablation ==="
pixi run -e parnet-dev-cu12 python scripts/train_cnn_only.py --run-id ablation_cnn_only_v1 --gpu 0

echo "=== [4/4] Auswertung aller drei Modelle -> results/new/ ==="
pixi run -e parnet-dev-cu12 python scripts/evaluate_new_models.py \
  --standard-run-id globalclip.standard.positional_v1 \
  --qlayer-run-id globalclip.qlayer.positional_v1 \
  --cnn-run-id ablation_cnn_only_v1 \
  --output-dir results/new

echo "=== ALLES FERTIG ==="
' > logs/train_and_eval_all.log 2>&1 &

## Experiment: Hybrid-Modell "CombiLayer" (CNN-only-Pfad + QLayer-Pfad)
## ─────────────────────────────────────────────────────────────────────────

# Idee: nicht mehr entscheiden muessen -- beide Signale (gewichtete Summe

# CNN-only werden, Phasen bleiben trotzdem trainierbar/auslesbar.
